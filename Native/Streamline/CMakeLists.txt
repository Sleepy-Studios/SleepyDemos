cmake_minimum_required(VERSION 3.24)
project(SleepyStreamline LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The bridge currently targets Windows x64.")
endif()
set(UNITY_PLUGIN_API "" CACHE PATH "Unity Editor/Data/PluginAPI")
set(STREAMLINE_SDK "" CACHE PATH "Verified Streamline SDK root")
set(VULKAN_HEADERS "" CACHE PATH "Vulkan-Headers root")
foreach(required "${UNITY_PLUGIN_API}/IUnityGraphicsD3D12.h" "${STREAMLINE_SDK}/include/sl.h" "${VULKAN_HEADERS}/include/vulkan/vulkan.h")
    if(NOT EXISTS "${required}")
        message(FATAL_ERROR "Missing dependency: ${required}")
    endif()
endforeach()
add_library(GfxPluginSleepyStreamline SHARED src/Bridge.cpp)
target_compile_features(GfxPluginSleepyStreamline PRIVATE cxx_std_17)
target_compile_definitions(GfxPluginSleepyStreamline PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX VK_NO_PROTOTYPES)
target_include_directories(GfxPluginSleepyStreamline PRIVATE "${UNITY_PLUGIN_API}" "${STREAMLINE_SDK}/include" "${VULKAN_HEADERS}/include")
target_link_libraries(GfxPluginSleepyStreamline PRIVATE d3d12 dxgi wintrust crypt32)
if(MSVC)
    target_compile_options(GfxPluginSleepyStreamline PRIVATE /W4 /EHsc /utf-8)
endif()
